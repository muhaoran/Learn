## SQL 语句

```sql
-- [SQL 整体说明]

-- CTE 1: [说明]
WITH cte1 AS (
    SELECT 
        field1,
        field2
    FROM 
        table1
    WHERE 
        condition1
),

-- CTE 2: [说明]
cte2 AS (
    SELECT 
        field3,
        field4
    FROM 
        table2
    WHERE 
        condition2
)

-- 主查询: [说明]
SELECT 
    -- 维度字段
    cte1.field1,
    
    -- 指标字段
    COUNT(DISTINCT cte1.field2) as metric1,
    COUNT(DISTINCT cte2.field3) as metric2,
    
    -- 计算字段
    ROUND(COUNT(DISTINCT cte2.field3) * 100.0 / COUNT(DISTINCT cte1.field2), 2) as rate
FROM 
    cte1
LEFT JOIN 
    cte2
ON 
    cte1.field1 = cte2.field1
GROUP BY 
    cte1.field1
ORDER BY 
    cte1.field1;
```
